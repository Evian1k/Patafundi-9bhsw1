   206→// Error logs — staff with can_view_logs can view system errors
   207→router.get('/staff/error-logs', authRequired, requirePermission('can_view_logs'), asyncHandler(async (req, res) => {
   208→  const { query: q } = await import('./db.js');
   209→  const limit = Math.min(Number(req.query.limit) || 50, 200);
   210→  const resolved = req.query.resolved === 'true' ? 'true' : (req.query.resolved === 'false' ? 'false' : null);
   211→  const where = resolved ? `where resolved = ${resolved}` : '';
   212→  const result = await q(`select * from error_logs ${where} order by created_at desc limit $1`, [limit]);
   213→  res.json({ success: true, errors: result.rows });
   214→}));

   412→  }
   413→  const { plan = 'monthly', amount = 500, mpesaNumber, idempotencyKey } = req.body || {};
   414→  if (!mpesaNumber) {
   415→    return res.status(400).json({ success: false, message: 'M-Pesa number required for subscription payment' });
   416→  }
   417→  const { query } = await import('./db.js');
   418→  const { assertValidMpesaPhone, initiateStkPush } = await import('./services/mpesaService.js');
   419→  const normalizedPhone = assertValidMpesaPhone(mpesaNumber);
   420→  const key = idempotencyKey || crypto.randomUUID();
   421→
   422→  // Create subscription as 'pending' — not active until payment confirmed
   423→  const subResult = await query(
   424→    `insert into subscriptions (fundi_id, plan, amount, status, starts_at, expires_at)
   425→     values ($1, $2, $3, 'pending', now(), now() + interval '30 days')
   426→     returning *`,
   427→    [req.user.id, plan, amount],
   428→  );
   429→
   430→  // Initiate M-Pesa STK push for the subscription fee

   437→    });
   438→
   439→    // Store checkout_request_id on the subscription for webhook matching
   440→    await query(
   441→      `update subscriptions set metadata = jsonb_set(coalesce(metadata, '{}'::jsonb), '{checkout_request_id}', $2::text)
   442→       where id = $1`,
   443→      [subResult.rows[0].id, JSON.stringify(daraja.CheckoutRequestID)],
   444→    );
   445→
   446→    res.status(202).json({
   447→      success: true,
   448→      subscriptionId: subResult.rows[0].id,

   450→      message: 'Subscription payment initiated. Complete the M-Pesa prompt to activate.',
   451→    });
   452→  } catch (err) {

   455→    res.status(502).json({
