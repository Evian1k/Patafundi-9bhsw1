     9→export class ApiError extends Error {
    10→  status: number;
    11→  meta?: unknown;
    12→  constructor(message: string, status = 0, meta?: unknown) {
    13→    super(message);
    14→    this.name = 'ApiError';
    15→    this.status = status;
    16→    this.meta = meta;
    17→  }

    20→const UNAVAILABLE_MSG = 'Service temporarily unavailable. Please try again.';
    21→
    22→async function sleep(ms: number) {
    23→  return new Promise((resolve) => setTimeout(resolve, ms));
    24→}
    25→
    26→function readCookie(name: string): string | null {
    27→  if (typeof document === 'undefined') return null;
    28→  const match = document.cookie.match(new RegExp(`(?:^|; )${name.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}=([^;]*)`));
    29→  return match ? decodeURIComponent(match[1]) : null;
    30→}
    31→
    32→class ApiClient {
    33→  token: string | null;
    34→
    35→  constructor() {
    36→    this.token = typeof localStorage !== 'undefined' ? localStorage.getItem('auth_token') : null;
    37→  }
    38→
    39→  setToken(token: string | null) {
    40→    this.token = token;
    41→    if (typeof localStorage !== 'undefined') {
    42→      if (token) localStorage.setItem('auth_token', token);
    43→      else localStorage.removeItem('auth_token');
    44→    }
    45→    if (!token) clearAuthSession();
    46→  }
    47→
    48→  private syncTokenFromStorage(): string | null {
    49→    if (typeof localStorage !== 'undefined') {
    50→      this.token = localStorage.getItem('auth_token');
    51→    }
    52→    return this.token;
    53→  }
    54→
    55→  private getHeaders(includeAuth = true): Record<string, string> {
    56→    const h: Record<string, string> = { 'Content-Type': 'application/json' };
    57→    const token = this.syncTokenFromStorage();
    58→    if (includeAuth && token) h['Authorization'] = `Bearer ${token}`;
    59→    const csrf = readCookie('csrf_token');
    60→    if (csrf) h['X-CSRF-Token'] = csrf;
    61→    return h;
    62→  }
    63→
    64→  async request(
    65→    endpoint: string,
    66→    options: RequestInit & { includeAuth?: boolean } = {},
    67→    retries = 1,
    68→  ): Promise<unknown> {
    69→    if (!isApiConfigured()) throw new ApiError(UNAVAILABLE_MSG, 0);
    70→
    71→    const { includeAuth = true, ...fetchOpts } = options;
    72→    const url = buildApiUrl(endpoint);
    73→
    74→    const config: RequestInit = {
    75→      ...fetchOpts,
    76→      credentials: 'include',
    77→      headers: {
    78→        ...this.getHeaders(includeAuth),
    79→        ...(fetchOpts.headers as Record<string, string> ?? {}),
    80→      },
    81→    };
    82→
    83→    if (config.body && typeof config.body === 'object' && !(config.body instanceof FormData)) {
    84→      config.body = JSON.stringify(config.body);
    85→    }
    86→    // Remove Content-Type for FormData (browser sets it with boundary)
    87→    if (config.body instanceof FormData) {
    88→      delete (config.headers as Record<string, string>)['Content-Type'];
    89→    }
    90→
    91→    for (let attempt = 0; attempt <= retries; attempt++) {
    92→      let response: Response;
    93→      try {
    94→        const controller = new AbortController();
    95→        const tId = setTimeout(() => controller.abort(), 30_000);
    96→        response = await fetch(url, { ...config, signal: controller.signal });
    97→        clearTimeout(tId);
    98→      } catch (e) {
    99→        if ((e as Error)?.name === 'AbortError') {
   100→          if (attempt < retries) { await sleep(1000); continue; }
   101→          throw new ApiError('Request timed out. Please check your connection.', 0);
   102→        }
   103→        if (attempt < retries) { await sleep(1000 * 2 ** attempt); continue; }
   104→        throw new ApiError('Unable to connect. Please check your internet connection.', 0);
   105→      }
   106→
   107→      if (response.status >= 500 && attempt < retries) {
   108→        await sleep(1000 * 2 ** attempt);
   109→        continue;
   110→      }
   111→
   112→      if (response.status === 401 && this.token && attempt === 0) {
   113→        // JWT expired — try to refresh once, then retry the original request
   114→        try {
   115→          const refreshed = await this.refreshToken();
   116→          if (refreshed) {
   117→            config.headers = config.headers || {};
   118→            (config.headers as Record<string, string>)['Authorization'] = `Bearer ${this.token}`;
   119→            continue; // retry with new token
   120→          }
   121→        } catch {
   122→          // refresh failed — clear token, redirect to login
   123→          this.setToken(null);
   124→          if (typeof window !== 'undefined' && !window.location.pathname.includes('/auth')) {
   125→            window.location.href = '/auth';
   126→          }
   127→        }
   128→      }
   129→
   130→      // 403 = Forbidden (permission denied) — do NOT retry, just throw.
   131→      // Refreshing the token won't help because the user's role/permissions
   132→      // haven't changed. Let the caller handle the 403 gracefully.
   133→
   134→      if (!response.ok) {
   135→        const errorBody = await response.json().catch(() => ({ message: response.statusText })) as { message?: string };
   136→        throw new ApiError(errorBody?.message || response.statusText || 'Request failed', response.status);
   137→      }
   138→
   139→      return response.json();
   140→    }
   141→
   142→    throw new ApiError(UNAVAILABLE_MSG, 0);
   143→  }

   145→  // ── Token refresh ─────────────────────────────────────────────────────
   146→  /** Attempts to refresh the JWT access token using the httpOnly refresh cookie. */
   147→  async refreshToken(): Promise<boolean> {
   148→    try {
   149→      const url = buildApiUrl('/auth/refresh');
   150→      const response = await fetch(url, {
   151→        method: 'POST',
   152→        credentials: 'include',
   153→        headers: { 'Content-Type': 'application/json' },
   154→      });
   155→      if (!response.ok) return false;
   156→      const data = await response.json();
   157→      if (data?.token) {
   158→        this.setToken(data.token);
   159→        return true;
   160→      }
   161→      return false;
   162→    } catch {
   163→      return false;
   164→    }
   165→  }

   547→  async uploadDisputeEvidence(disputeId: string, formData: FormData) {
   548→    if (!isApiConfigured()) throw new ApiError(UNAVAILABLE_MSG, 0);
   549→    const url = buildApiUrl(`/disputes/${disputeId}/evidence`);
   550→    const response = await fetch(url, {
   551→      method: 'POST',
   552→      credentials: 'include',
   553→      headers: {
   554→        Authorization: `Bearer ${this.token ?? ''}`,
   555→        ...(readCookie('csrf_token') ? { 'X-CSRF-Token': readCookie('csrf_token') as string } : {}),
   556→      },
   557→      body: formData,
   558→    });
   559→    if (!response.ok) {
   560→      const e = await response.json().catch(() => ({ message: response.statusText })) as { message?: string };
   561→      throw new ApiError(e.message || 'Upload failed', response.status);
   562→    }
   563→    return response.json();
   564→  }
